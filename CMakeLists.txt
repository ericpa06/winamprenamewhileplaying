cmake_minimum_required(VERSION 3.21)
project(WinampFileShareUnlocker LANGUAGES C CXX)
enable_testing()
if(WIN32)
  include(FetchContent)
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "Build MinHook statically" FORCE)
  FetchContent_Declare(minhook GIT_REPOSITORY https://github.com/TsudaKageyu/minhook.git GIT_TAG v1.3.4 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(minhook)
  add_library(gen_fileunlock SHARED src/plugin.cpp)
  target_include_directories(gen_fileunlock PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include")
  target_link_libraries(gen_fileunlock PRIVATE minhook user32)
  target_compile_features(gen_fileunlock PRIVATE cxx_std_17)
  if(MSVC)
    target_compile_options(gen_fileunlock PRIVATE /W4 /permissive- /EHsc /utf-8)
  endif()
  set_target_properties(gen_fileunlock PROPERTIES PREFIX "" OUTPUT_NAME "gen_fileunlock" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
  foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_target_properties(gen_fileunlock PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/bin" LIBRARY_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/bin")
  endforeach()
  add_executable(share_semantics_test tests/share_semantics_windows.cpp)
  target_compile_features(share_semantics_test PRIVATE cxx_std_17)
  add_test(NAME share_semantics_test COMMAND share_semantics_test)
endif()
add_executable(path_scope_test tests/path_scope_test.cpp)
target_include_directories(path_scope_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(path_scope_test PRIVATE cxx_std_17)
add_test(NAME path_scope_test COMMAND path_scope_test)
